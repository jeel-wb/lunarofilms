    if (e.ctrlKey && e.key === "U") e.preventDefault();
});

    if (window.outerWidth - window.innerWidth > 100) {
        document.body.style.filter = "blur(15px)";

// 1. MUSIC CONTROL
const music = document.getElementById("bgMusic");
const record = document.getElementById("record");
const playIcon = document.getElementById("playIcon");
const statusText = document.getElementById("statusText");

// 🔥 Restore Time + State on Load
window.addEventListener("load", () => {
    const savedTime = localStorage.getItem("musicTime");
    const savedState = localStorage.getItem("musicState");

    if (savedTime) {
        music.currentTime = parseFloat(savedTime);
    }

    if (savedState === "paused") {
        music.pause();
        record.style.animationPlayState = 'paused';
        playIcon.classList.replace("fa-pause", "fa-play");
        statusText.innerText = "PAUSED";
    } else {
        // default = playing
        music.play().catch(() => {});
        record.style.animationPlayState = 'running';
        playIcon.classList.replace("fa-play", "fa-pause");
        statusText.innerText = "PLAYING";
    }
});

// 🔥 First Click Fix (Autoplay policy)
window.addEventListener('click', () => {
    if (music.paused && statusText.innerText === "PLAYING") {
        music.play().catch(() => {});
        record.style.animationPlayState = 'running';
    }
}, { once: true });

// 🔥 Save current time continuously
music.addEventListener("timeupdate", () => {
    localStorage.setItem("musicTime", music.currentTime);
});

// 🔥 Toggle Play/Pause + Save State
document.getElementById("musicToggle").onclick = (e) => {
    e.stopPropagation();

    if (music.paused) {
        music.play();
        record.style.animationPlayState = 'running';
        playIcon.classList.replace("fa-play", "fa-pause");
        statusText.innerText = "PLAYING";
        localStorage.setItem("musicState", "playing");
    } else {
        music.pause();
        record.style.animationPlayState = 'paused';
        playIcon.classList.replace("fa-pause", "fa-play");
        statusText.innerText = "PAUSED";
        localStorage.setItem("musicState", "paused");
    }
};
